<script>
    // Remove/esconde o iframe do Netlify
    (function hideNetlifyBadge() {
        function removeBadge() {
            // Tenta esconder pelo ID
            var iframe = document.getElementById('nl-badge-frame');
            if (iframe) {
                iframe.style.display = 'none';
                iframe.style.visibility = 'hidden';
                iframe.style.opacity = '0';
                iframe.remove(); // Remove completamente do DOM
            }
            
            // Tenta esconder por seletor CSS genérico
            var badges = document.querySelectorAll('iframe[title*="Netlify"], iframe[srcdoc*="Netlify"]');
            badges.forEach(function(badge) {
                badge.style.display = 'none';
                badge.remove();
            });
        }

        // Executa imediatamente
        removeBadge();

        // Executa após DOM carregado
        if (document.readyState === 'loading') {
            document.addEventListener('DOMContentLoaded', removeBadge);
        }

        // Executa após página carregar completamente
        window.addEventListener('load', function() {
            setTimeout(removeBadge, 100);
            setTimeout(removeBadge, 500);
            setTimeout(removeBadge, 1500);
        });

        // Usa MutationObserver para detectar quando o iframe for injetado
        if (window.MutationObserver) {
            var observer = new MutationObserver(function(mutations) {
                mutations.forEach(function(mutation) {
                    mutation.addedNodes.forEach(function(node) {
                        if (node.tagName === 'IFRAME' && 
                            (node.id === 'nl-badge-frame' || 
                             node.title.indexOf('Netlify') !== -1)) {
                            node.style.display = 'none';
                            setTimeout(function() { node.remove(); }, 10);
                        }
                    });
                });
            });

            observer.observe(document.body || document.documentElement, {
                childList: true,
                subtree: true
            });
        }
    })();
</script>
